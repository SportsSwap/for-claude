15-minute level breakout

15-minute chart · NQ1! or MNQ1! Failed on 2013–2021

Trades the first 15-minute close through a level made of three swing points, long or short, between 8am and 4pm New York time. Stop beyond the last swing, target equal to the risk, out after 96 bars.

Lab, Aug 2024 to Nov 2026: NQ 941 trades, Sharpe 0.72 · ES 993 trades, Sharpe 0.59. Older years, 2013 to Sep 2021: NQ 3,383 trades, Sharpe 0.17 · ES 3,290 trades, Sharpe −0.51 (lost money).

//@version=6
// Price Action Lab survivor: C-breakout-00051
// Survived the lab's sealed final test (2024-08-01 to 2026-11-06, Dukascopy CFD data):
// NQ 941 trades, +$112,064, Sharpe 0.72; ES 993 trades, +$45,704, Sharpe 0.59.
// Dials are fixed by the lab. Changing any value tests a different, untested version.
// Costs as in the lab: $4 per round trip, 1 tick slippage, limits need 1 tick through.
// Differences from the lab: TradingView fills on chart bars, not 1-minute bars, and on a
// bar that hits both stop and target it may not assume the stop first (the lab does).
// Margin is not modelled (one contract, as in the lab).
// Paper trade before risking money.
strategy("PA Lab 15m Level Breakout", shorttitle="PALab Breakout", overlay=true,
     process_orders_on_close=true, pyramiding=0, default_qty_type=strategy.fixed, default_qty_value=1,
     commission_type=strategy.commission.cash_per_contract, commission_value=2.0, slippage=1,
     backtest_fill_limits_assumption=1, margin_long=0, margin_short=0, max_bars_back=600)

// ---------------------------------------------------------------- dials (from the lab)
int   SWING     = 3       // swing strength: bars each side
float TOL_ATR   = 0.25    // swings within this many ATR form one level
int   TOUCHES   = 3       // swings a level needs
int   LIFE      = 500     // bars a level lives after its last swing
float STOP_BUF  = 0.1     // stop buffer beyond the last swing, in ATR
float TARGET_R  = 1.0     // target in multiples of risk
int   TIME_BARS = 96      // exit after this many 15m bars
int   SESS_FROM = 480     // entries only when the bar closes 08:00 ...
int   SESS_TO   = 960     // ... to 16:00 New York time

if timeframe.period != "15"
    runtime.error("Put this strategy on a 15-minute chart (NQ1! or ES1!).")

float atrPrior = ta.atr(14)[1]          // Wilder ATR through the previous bar only

// ---------------------------------------------------------------- swings (confirmed SWING bars later)
// The bar SWING bars ago is a swing high when its high beats the SWING highs before it
// and is at least as high as the SWING highs after it (no loops: built-ins only).
float maxBefore = ta.highest(high, SWING)[SWING + 1]
float maxAfter  = ta.highest(high, SWING)
float minBefore = ta.lowest(low, SWING)[SWING + 1]
float minAfter  = ta.lowest(low, SWING)
bool newHigh = bar_index > 2 * SWING and high[SWING] > maxBefore and high[SWING] >= maxAfter
bool newLow  = bar_index > 2 * SWING and low[SWING] < minBefore and low[SWING] <= minAfter
var float lastSwingHigh = na
var float lastSwingLow  = na

// ---------------------------------------------------------------- levels: clusters of swings
var array<float> clSum  = array.new<float>()
var array<int>   clN    = array.new<int>()
var array<int>   clLast = array.new<int>()
var array<float> lvPrice = array.new<float>()
var array<int>   lvBorn  = array.new<int>()

var int nSwings = 0
var int nLevelsMade = 0
var int nBreaks = 0

// new swings this bar (a bar can confirm a swing high and a swing low)
array<float> fresh = array.new<float>()
if newHigh
    lastSwingHigh := high[SWING]
    array.push(fresh, high[SWING])
if newLow
    lastSwingLow := low[SWING]
    array.push(fresh, low[SWING])

if array.size(fresh) > 0 and not na(atrPrior)
    for p in fresh
        nSwings += 1
        int best = -1
        float bestD = 1.0e18
        int n = array.size(clSum)
        if n > 0
            for i = 0 to n - 1
                if bar_index - array.get(clLast, i) <= LIFE
                    float d = math.abs(p - array.get(clSum, i) / array.get(clN, i))
                    if d <= TOL_ATR * atrPrior and d < bestD
                        best := i
                        bestD := d
        if best < 0
            array.push(clSum, p)
            array.push(clN, 1)
            array.push(clLast, bar_index)
            best := array.size(clSum) - 1
        else
            array.set(clSum, best, array.get(clSum, best) + p)
            array.set(clN, best, array.get(clN, best) + 1)
            array.set(clLast, best, bar_index)
        if array.get(clN, best) >= TOUCHES
            array.push(lvPrice, array.get(clSum, best) / array.get(clN, best))
            array.push(lvBorn, bar_index)
            nLevelsMade += 1
            array.remove(clSum, best)
            array.remove(clN, best)
            array.remove(clLast, best)
        int m = array.size(clSum)
        if m > 0
            for i = m - 1 to 0
                if bar_index - array.get(clLast, i) > LIFE
                    array.remove(clSum, i)
                    array.remove(clN, i)
                    array.remove(clLast, i)

// ---------------------------------------------------------------- breaks: first close through a level
int breakDir = 0
float breakLevel = na
int nl = array.size(lvPrice)
if nl > 0
    for i = nl - 1 to 0
        if bar_index - array.get(lvBorn, i) > LIFE
            array.remove(lvPrice, i)
            array.remove(lvBorn, i)
nl := array.size(lvPrice)
if nl > 0
    for i = 0 to nl - 1
        float lv = array.get(lvPrice, i)
        if breakDir == 0 and bar_index > array.get(lvBorn, i)
            if close[1] < lv and close > lv
                breakDir := 1
                breakLevel := lv
            else if close[1] > lv and close < lv
                breakDir := -1
                breakLevel := lv

// ---------------------------------------------------------------- entries and exits
int closeMin = hour(time_close, "America/New_York") * 60 + minute(time_close, "America/New_York")
bool inSession = closeMin >= SESS_FROM and closeMin <= SESS_TO
var int entryBar = na

if strategy.position_size != 0 and not na(entryBar) and bar_index >= entryBar + TIME_BARS
    strategy.close_all(comment="time exit")
    entryBar := na

if strategy.position_size == 0 and breakDir != 0 and inSession and not na(atrPrior)
    if breakDir == 1 and not na(lastSwingLow)
        float stop = lastSwingLow - STOP_BUF * atrPrior
        float risk = close - stop
        if risk >= syminfo.mintick
            strategy.entry("Long", strategy.long)
            strategy.exit("Long exit", "Long", stop=stop, limit=close + TARGET_R * risk)
            entryBar := bar_index
    else if breakDir == -1 and not na(lastSwingHigh)
        float stop = lastSwingHigh + STOP_BUF * atrPrior
        float risk = stop - close
        if risk >= syminfo.mintick
            strategy.entry("Short", strategy.short)
            strategy.exit("Short exit", "Short", stop=stop, limit=close - TARGET_R * risk)
            entryBar := bar_index

if breakDir != 0
    nBreaks += 1

plotshape(breakDir == 1 and inSession, "Break up", shape.triangleup, location.belowbar, color.new(color.teal, 0), size=size.small)
plotshape(breakDir == -1 and inSession, "Break down", shape.triangledown, location.abovebar, color.new(color.red, 0), size=size.small)

// status label on the last bar: shows where the logic stops if nothing trades
var label status = na
if barstate.islast
    label.delete(status)
    status := label.new(bar_index, high, "swings " + str.tostring(nSwings) + " | levels made " + str.tostring(nLevelsMade) +
         " | live levels " + str.tostring(array.size(lvPrice)) + " | breaks " + str.tostring(nBreaks),
         style=label.style_label_down, color=color.new(color.gray, 20), textcolor=color.white, size=size.small)

Daily fair value gap

1-hour chart · NQ1! or MNQ1! Failed on 2013–2021

After each daily close, a three-day gap gets a limit order at its near edge, with the stop beyond the far edge and a target of three times the risk. The order lives, and any trade closes, after 96 hourly bars. Gaps formed on CPI, jobs-report or Fed days are skipped.

Lab, Aug 2024 to Nov 2026: NQ 50 trades, Sharpe 1.01 · ES 49 trades, Sharpe 0.84. Older years, 2013 to Sep 2021: NQ 175 trades, Sharpe 0.09 · ES 165 trades, Sharpe 0.12 (about break-even; 2020 was the only clearly winning year on NQ).

//@version=6
// Price Action Lab survivor: C-fair_value_gap-00364

// Dials are fixed by the lab. Changing any value tests a different, untested version.
// Costs as in the lab: $4 per round trip, 1 tick slippage, limits need 1 tick through.
// Differences from the lab: TradingView fills on chart bars, not 1-minute bars, and on a
// bar that hits both stop and target it may not assume the stop first (the lab does).
// Margin is not modelled (one contract, as in the lab).

Noise band (the post's rules), any symbol

5-minute chart · any stock, ETF or future Tested on QQQ only

Opening range (first 15 minutes), a noise band from the last 20 days' average distance from the open at each minute (x1.25), and a 5-day ATR, all worked out from the session set in the settings (default 9:30-16:00 New York, right for US stocks and ETFs). Buys or shorts when price closes outside the band and the range. Version menu: the post's rules as written, or the best version picked on 2013-2021 (30-minute checks, out when price crosses back through VWAP). Needs 20 days of history: the label on the last bar must read 20/20. On futures (NQ1!, MNQ1!) it trades a fixed number of contracts, set in the settings, and you should set Properties > Commission to about $2 a contract for NQ or $0.50 for MNQ.

Lab, QQQ itself, simple sums after costs: as written 2013-Sep 2021 +7.7% · 2022-Jul 2024 +36.7% · Aug 2024-Nov 2026 +12.6%; best from 2013-2021 +30.4% · +21.0% · +8.7%, worst drop 8.4%. On the S&P 500 (SPY) the same rules were flat or losing (best version -2.9% · +9.9% · -0.2%). Other symbols untested. Options on these signals lost money in the lab's estimate.

//@version=6
// PA Lab: noise band (the user's rules, decisions.md D64-D69). Works on any symbol.
// Chart: 5-minute (recommended) or any minute chart up to 30 minutes.
// Set the session below if the symbol is not a US stock/ETF (default 9:30-16:00 New York).
//
// Two versions:
//   "As written"            the rules from the post: checks every 10 minutes, VWAP filter, opening
//                           range + 0.3 ATR, out 15 minutes before the close.
//   "Best from 2013-2021"   picked on 2013-Sep 2021 only (Nasdaq-100): checks every 30 minutes,
//                           range + 0.15 ATR, no VWAP entry filter (it changed nothing), out when price closes back
//                           through VWAP or 0.5 ATR against, otherwise 15 minutes before the close.
// Tested in the lab on the Nasdaq-100 (QQQ): it made money in all three periods.
// On the S&P 500 (SPY) the same rules were about flat or losing. Any other symbol is untested.
// Buying 1-3 day options on these signals lost money in the lab's estimate in most years.

// Futures (NQ1!, MNQ1!, ES1!...): it trades a fixed number of contracts (setting below), because
// 100% of equity is less than one NQ contract and TradingView would round the order to zero.
// For futures also set Properties > Commission to about $2 per contract (NQ) or $0.50 (MNQ):
// the default 0.01% of trade value is right for stocks but far too high for futures.
strategy("PA Lab Noise Band", shorttitle="Noise Band", overlay=true,
     process_orders_on_close=true, pyramiding=0, default_qty_type=strategy.percent_of_equity,
     default_qty_value=100, commission_type=strategy.commission.percent, commission_value=0.01,
     slippage=1, margin_long=0, margin_short=0)

                stage := 3
